    {\Large\color{cs2inkSoft} #3}\\[8mm]
    {\color{#1}\rule{6cm}{2pt}}
  \end{center}
\end{frame}
}

    \item[] \textbf{\color{worldinvestigation}Investigation Bureau} -- a case bureau where the records don't always agree.
    \item[] \textbf{\color{worldstarship}Starship Log} -- a working ship under real operational pressure.
    \item[] \textbf{\color{worldbusiness}Small Business} -- a real business whose systems are buckling under growth.
  \end{itemize}

  earned. Overthinking this is unnecessary.
